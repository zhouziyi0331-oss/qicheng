## 📡 API接口

### 1. 生成晋级验证内容
```typescript
POST /api/level-up-validation/generate

Request Body:
{
  fromLevel: 0,
  toLevel: 1
}

Response:
{
  success: true,
  data: {
    level: 1,
    title: "第一次把自己卖出去",
    mentorMessage: "第一单做完了。我记得你接单那天...",
    question: {
      text: "这几个瞬间里，哪一个让你觉得「啊，好像我真的可以」？",
      options: [
        {
          id: "A",
          text: "卡住之后，自己想办法想通了",
          value: "problem_solving"
        },
        // ...更多选项
      ]
    },
    mentorClosing: "好。我记住了。走吧，Lv.1了。",
    timeline: {  // 仅Lv5有
      firstProjectDate: "2026-01-15",
      todayDate: "2026-07-22"
    }
  }
}
```

### 2. 提交晋级验证答案
```typescript
POST /api/level-up-validation/submit

Request Body:
{
  level: 1,
  questionId: "main",
  selectedOption: "problem_solving"
}

Response:
{
  success: true,
  data: {
    success: true,
    newLevel: 1,
    message: "恭喜晋级！"
  }
}
```

### 3. 获取晋级历史
```typescript
GET /api/level-up-validation/history

Response:
{
  success: true,
  data: {
    history: [
      {
        level: 1,
        achievedAt: "2026-01-15T10:30:00Z",
        validationAnswer: "problem_solving"
      }
    ]
  }
}
```

### 页面核心逻辑
```typescript
import { View, Text, Radio, RadioGroup, Button } from '@tarojs/components'
import Taro, { useLoad } from '@tarojs/taro'
import { useState } from 'react'

export default function LevelUpValidation() {
  const [validation, setValidation] = useState<any>(null)
  const [selectedOption, setSelectedOption] = useState('')
  const [loading, setLoading] = useState(false)
  const [showClosing, setShowClosing] = useState(false)
  
  useLoad((options) => {
    const toLevel = parseInt(options.toLevel || '1')
    loadValidation(toLevel)
  })

  // 加载验证内容
  const loadValidation = async (toLevel: number) => {
    try {
      const token = Taro.getStorageSync('token')
      const res = await Taro.request({
        url: 'http://localhost:3000/api/level-up-validation/generate',
        method: 'POST',
        header: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json'
        },
        data: {
          fromLevel: toLevel - 1,
          toLevel: toLevel
        }
      })

      if (res.statusCode === 200 && res.data.success) {
        setValidation(res.data.data)
      }
    } catch (error) {
      console.error('加载晋级验证失败:', error)
      Taro.showToast({
        title: '加载失败',
        icon: 'none'
      })
    }
  }

  // 提交答案
  const handleSubmit = async () => {
    if (!selectedOption) {
      Taro.showToast({
        title: '请选择一个选项',
        icon: 'none'
      })
      return
    }

    setLoading(true)
    
    try {
      const token = Taro.getStorageSync('token')
      const res = await Taro.request({
        url: 'http://localhost:3000/api/level-up-validation/submit',
        method: 'POST',
        header: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json'
        },
        data: {
          level: validation.level,
          questionId: 'main',
          selectedOption: selectedOption
        }
      })

      if (res.statusCode === 200 && res.data.success) {
        // 显示导师结束语
        setShowClosing(true)
        
        // 2秒后跳转
        setTimeout(() => {
          Taro.redirectTo({
            url: '/pages/profile/index'
          })
        }, 2000)
      }
    } catch (error) {
      console.error('提交失败:', error)
      Taro.showToast({
        title: '提交失败',
        icon: 'none'
      })
    } finally {
      setLoading(false)
    }
  }

  if (!validation) {
    return <View>加载中...</View>
  }

  // 显示结束语
  if (showClosing) {
    return (
      <View className="level-up-closing">
        <Text className="mentor-avatar">🐱</Text>
        <Text className="closing-text">{validation.mentorClosing}</Text>
      </View>
    )
  }

  return (
    <View className="level-up-validation">
      {/* 头部 */}
      <View className="header">
        <View className="cat-avatar">🐱</View>
        <Text className="mentor-name">启程小猫</Text>
        <Text className="level-progress">
          Lv.{validation.level - 1} → Lv.{validation.level}
        </Text>
      </View>

      {/* 导师的话 */}
      <View className="mentor-message">
        <Text className="message-text">{validation.mentorMessage}</Text>
      </View>

      {/* 问题 */}
      <View className="question-section">
        <Text className="question-text">{validation.question.text}</Text>
        
        <RadioGroup onChange={(e) => setSelectedOption(e.detail.value)}>
          {validation.question.options.map((option: any) => (
            <View key={option.id} className="option-item">
              <Radio value={option.value} checked={selectedOption === option.value} />
              <Text className="option-text">
                {option.id}. {option.text}
              </Text>
            </View>
          ))}
        </RadioGroup>
      </View>

      {/* Lv5的时间线 */}
      {validation.timeline && (
        <View className="timeline">
          <Text>从 {validation.timeline.firstProjectDate}</Text>
          <Text>到 {validation.timeline.todayDate}</Text>
        </View>
      )}

      {/* 确认按钮 */}
      <View className="submit-section">
        <Button 
          className="submit-btn"
          onClick={handleSubmit}
          disabled={!selectedOption || loading}
        >
          {loading ? '提交中...' : '确认选择'}
        </Button>
      </View>
    </View>
  )
}
```

## 🔄 触发时机

### 方式1：完成项目后自动检查
```typescript
// 在项目完成接口返回后
if (response.data.shouldLevelUp) {
  Taro.showModal({
    title: '恭喜！',
    content: '你已经满足晋级条件，要现在进行晋级验证吗？',
    success: (res) => {
      if (res.confirm) {
        Taro.navigateTo({
          url: `/pages/level-up-validation/index?toLevel=${response.data.nextLevel}`
        })
      }
    }
  })
}
```


## 🎯 核心特点

1. **个性化内容**：每个人看到的导师的话都不同，基于真实数据
2. **只有1个问题**：不是5道题，导师只问最核心的
3. **有"不知道"选项**：允许学生诚实
4. **简洁有力的结束语**："走，Lv.X"
5. **无需打字**：全部选择题，降低压力
