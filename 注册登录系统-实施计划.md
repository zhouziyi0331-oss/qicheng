import { Redis } from 'ioredis'

// 开发环境使用固定验证码，生产环境调用真实短信API
const MOCK_CODE = '123456'
const isDev = process.env.NODE_ENV === 'development'

class SMSService {
  private redis: Redis

  constructor() {
    this.redis = new Redis({
      host: process.env.REDIS_HOST || 'localhost',
      port: parseInt(process.env.REDIS_PORT || '6379')
    })
  }

  /**
   * 发送验证码
   */
  async sendVerifyCode(phone: string, type: 'login' | 'register'): Promise<void> {
    // 检查发送频率限制
    const rateLimitKey = `sms:rate:${phone}`
    const lastSent = await this.redis.get(rateLimitKey)
    
    if (lastSent) {
      throw new Error('验证码发送太频繁，请60秒后再试')
    }

    let code: string
    
    if (isDev) {
      // 开发环境：使用固定验证码
      code = MOCK_CODE
      console.log(`📱 [开发环境] 验证码: ${code}`)
    } else {
      // 生产环境：调用真实短信服务
      code = this.generateCode()
      await this.sendRealSMS(phone, code, type)
    }

    // 存储验证码（5分钟有效期）
    const codeKey = `sms:code:${phone}:${type}`
    await this.redis.setex(codeKey, 300, code)

    // 设置发送频率限制（60秒）
    await this.redis.setex(rateLimitKey, 60, Date.now().toString())
  }

  /**
   * 验证验证码
   */
  async verifyCode(phone: string, code: string, type: 'login' | 'register'): Promise<boolean> {
    const codeKey = `sms:code:${phone}:${type}`
    const savedCode = await this.redis.get(codeKey)

    if (!savedCode) {
      throw new Error('验证码已过期或不存在')
    }

    if (savedCode !== code) {
      throw new Error('验证码错误')
    }

    // 验证成功后删除验证码
    await this.redis.del(codeKey)
    return true
  }

  /**
   * 生成6位数字验证码
   */
  private generateCode(): string {
    return Math.floor(100000 + Math.random() * 900000).toString()
  }

#### 1.2 账号类型验证

**更新文件**: `backend/src/services/auth.service.ts`

```typescript
class AuthService {
  /**
   * 检查手机号是否已注册
   */
  async checkPhoneRegistered(phone: string): Promise<{
    registered: boolean
    accountType?: 'student' | 'enterprise'
  }> {
    // 先检查学生表
    const student = await prisma.student.findUnique({
      where: { phone }
    })

    if (student) {
      return {
        registered: true,
        accountType: 'student'
      }
    }

    // 再检查企业表
    const enterprise = await prisma.enterprise.findUnique({
      where: { phone }
    })

    if (enterprise) {
      return {
        registered: true,
        accountType: 'enterprise'
      }
    }

    return {
      registered: false
    }
  }

  /**
   * 手机号注册
   */
  async registerByPhone(
    phone: string,
    code: string,
    accountType: 'student' | 'enterprise',
    nickname?: string
  ) {
    // 1. 验证验证码
    await smsService.verifyCode(phone, code, 'register')

    // 2. 检查手机号是否已被注册
    const existing = await this.checkPhoneRegistered(phone)
    
    if (existing.registered) {
      throw new Error(`该手机号已注册为${existing.accountType === 'student' ? '学生' : '企业'}账号`)
    }

    // 3. 创建账号
    if (accountType === 'student') {
      const student = await prisma.student.create({
        data: {
          phone,
          nickname: nickname || `学生${phone.slice(-4)}`,
          account_type: 'student',
          hasCompletedOnboarding: false // 标记未完成测评
        }
      })

      return {
        userId: student.id,
        accountType: 'student',
        needsOnboarding: true
      }
    } else {
      const enterprise = await prisma.enterprise.create({
        data: {
          phone,
          name: nickname || `企业${phone.slice(-4)}`,
          account_type: 'enterprise'
        }
      })

      return {
        userId: enterprise.id,
        accountType: 'enterprise',
        needsOnboarding: false
      }
    }
  }

  /**
   * 手机号登录
   */
  async loginByPhone(phone: string, code: string) {
    // 1. 验证验证码
    await smsService.verifyCode(phone, code, 'login')

    // 2. 查找账号
    const existing = await this.checkPhoneRegistered(phone)

    if (!existing.registered) {
      throw new Error('该手机号未注册，请先注册')
    }

    // 3. 获取账号信息
    let user: any
    if (existing.accountType === 'student') {
      user = await prisma.student.findUnique({
        where: { phone }
      })
    } else {
      user = await prisma.enterprise.findUnique({
        where: { phone }
      })
    }

    // 4. 生成token
    const token = this.generateToken(user.id, existing.accountType)

    return {
      token,
      userId: user.id,
      accountType: existing.accountType,
      needsOnboarding: user.hasCompletedOnboarding === false
    }
  }

  /**
   * 微信登录
   */
  async loginByWechat(code: string, accountType: 'student' | 'enterprise') {
    // 1. 通过code换取openid
    const { openid, unionid } = await this.getWechatOpenId(code)

    // 2. 查找是否已绑定
    let user: any
    if (accountType === 'student') {
      user = await prisma.student.findFirst({
        where: { wechatOpenId: openid }
      })
    } else {
      user = await prisma.enterprise.findFirst({
        where: { wechatOpenId: openid }
      })
    }

    if (user) {
      // 已绑定，直接登录
      const token = this.generateToken(user.id, accountType)
      return {
        token,
        userId: user.id,
        accountType,
        needsOnboarding: user.hasCompletedOnboarding === false,
        needsBindPhone: !user.phone
      }
    } else {
      // 未绑定，返回openid让前端引导绑定手机号
      return {
        needsBindPhone: true,
        openid,
        unionid
      }
    }
  }

  /**
   * 获取微信OpenID
   */
  private async getWechatOpenId(code: string): Promise<{
    openid: string
    unionid?: string
  }> {
    const appId = process.env.WECHAT_APP_ID
    const appSecret = process.env.WECHAT_APP_SECRET

    const response = await fetch(
      `https://api.weixin.qq.com/sns/jscode2session?appid=${appId}&secret=${appSecret}&js_code=${code}&grant_type=authorization_code`
    )

    const data = await response.json()

    if (data.errcode) {
      throw new Error(`微信登录失败: ${data.errmsg}`)
    }

    return {
      openid: data.openid,
      unionid: data.unionid
    }
  }

  /**
   * 绑定微信和手机号
   */
  async bindWechatAndPhone(
    openid: string,
    phone: string,
    code: string,
    accountType: 'student' | 'enterprise'
  ) {
    // 1. 验证验证码
    await smsService.verifyCode(phone, code, 'register')

    // 2. 检查手机号是否已注册
    const existing = await this.checkPhoneRegistered(phone)

    if (existing.registered && existing.accountType !== accountType) {
      throw new Error(`该手机号已注册为${existing.accountType === 'student' ? '学生' : '企业'}账号`)
    }

    // 3. 创建或更新账号
    let user: any
    if (accountType === 'student') {
      user = await prisma.student.upsert({
        where: { phone },
        create: {
          phone,
          wechatOpenId: openid,
          nickname: `学生${phone.slice(-4)}`,
          account_type: 'student',
          hasCompletedOnboarding: false
        },
        update: {
          wechatOpenId: openid
        }
      })
    } else {
      user = await prisma.enterprise.upsert({
        where: { phone },
        create: {
          phone,
          wechatOpenId: openid,
          name: `企业${phone.slice(-4)}`,
          account_type: 'enterprise'
        },
        update: {
          wechatOpenId: openid
        }
      })
    }

    const token = this.generateToken(user.id, accountType)

    return {
      token,
      userId: user.id,
      accountType,
      needsOnboarding: user.hasCompletedOnboarding === false
    }
  }

  private generateToken(userId: string, accountType: string): string {
    // 使用JWT生成token
    return jwt.sign(
      { userId, accountType },
      process.env.JWT_SECRET || 'your-secret-key',
      { expiresIn: '30d' }
    )
  }
}

export const authService = new AuthService()
```
